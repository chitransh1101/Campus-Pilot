# Core dependencies to run the CampusPilot API.
#
# Deliberately minimal and dependency-light: password hashing uses the
# Python standard library (hashlib's PBKDF2, see app/auth/auth.py) instead
# of passlib+bcrypt — that combo has a well-known runtime crash with recent
# bcrypt releases (passlib probing a `bcrypt.__about__` attribute bcrypt
# removed) that "works after pip install" doesn't catch, it only shows up
# the first time someone actually logs in. JWTs use PyJWT instead of
# python-jose for the same reason: fewer native/C-extension dependencies
# that can fail to build or version-conflict on a fresh machine.
fastapi==0.128.8
uvicorn==0.39.0
sqlalchemy==2.0.52
pydantic==2.13.4
pyjwt==2.12.1
python-multipart==0.0.20

# Only needed if you switch DATABASE_URL to MySQL instead of the SQLite
# default (see app/database/db.py).
pymysql==1.2.0

# Real attendance-photo OCR (app/ai/attendance_ocr.py). These two alone are
# NOT enough — pytesseract is a thin wrapper around the actual Tesseract-OCR
# program, which is a separate system install (not a pip package):
#   Windows : https://github.com/UB-Mannheim/tesseract/wiki  (installer)
#   macOS   : brew install tesseract
#   Linux   : apt-get install tesseract-ocr
# Without Tesseract installed, /teacher/attendance/ocr still works — it just
# falls back to simulated demo data (see attendance_ocr.py's caller).
opencv-python-headless==4.11.0.86
pytesseract==0.3.13
